{{- /* Highwire Press citation tags. This is the metadata format Google Scholar reads to index a working paper and to link it to its PDF; without it a paper page is just another HTML document to the crawler. Emitted on individual paper pages only. */ -}} {{- if and .IsPage (eq .Section "papers") }} {{- $authors := .Params.author | default site.Params.author }} {{- if reflect.IsSlice $authors }} {{- range $authors }} {{- end }} {{- else }} {{- with $authors }} {{- end }} {{- end }} {{- with site.Params.identity.affiliation }} {{- end }} {{- if not .Date.IsZero }} {{- end }} {{- with .Params.summary }} {{- end }} {{- /* journalName is the bare title; journal may carry a status prefix such as "Accepted, ...", which must never reach citation_journal_title. */ -}} {{- with (.Params.journalName | default .Params.journal) }} {{- end }} {{- with .Params.doi }} {{- end }} {{- with .Params.issn }} {{- end }} {{- with .Params.volume }} {{- end }} {{- with .Params.issue }} {{- end }} {{- with .Params.firstpage }} {{- end }} {{- with .Params.lastpage }} {{- end }} {{- /* pdf may be a file in static/ or a URL at the publisher; absURL must not be applied to one that is already absolute. */ -}} {{- with .Params.pdf }} {{- $u := . }}{{ if not (or (hasPrefix . "http://") (hasPrefix . "https://")) }}{{ $u = absURL . }}{{ end }} {{- end }} {{- range .Params.tags }} {{- end }} {{- end }}
