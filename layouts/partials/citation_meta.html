Emitted on individual paper pages only. */ -}} {{- if and .IsPage (eq .Section "papers") }} {{- $authors := .Params.author | default site.Params.author }} {{- if reflect.IsSlice $authors }} {{- range $authors }} {{- end }} {{- else }} {{- with $authors }} {{- end }} {{- end }} {{- with site.Params.identity.affiliation }} {{- end }} {{- if not .Date.IsZero }} {{- end }} {{- with .Params.summary }} {{- end }} {{- with .Params.journal }} {{- end }} {{- with .Params.doi }} {{- end }} {{- with .Params.pdf }} {{- end }} {{- range .Params.tags }} {{- end }} {{- end }}
